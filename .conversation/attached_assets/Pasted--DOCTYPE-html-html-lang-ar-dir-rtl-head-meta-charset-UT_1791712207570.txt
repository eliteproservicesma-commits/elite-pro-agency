// GSAP Hero Reveal Animation
gsap.registerPlugin(CustomEase);
const customEaseIn = CustomEase.create('custom-ease-in', '0.52, 0.00, 0.48, 1.00');

window.addEventListener('DOMContentLoaded', () => {
  const tl = gsap.timeline();
  tl.fromTo('.header', {y: '-100%'}, {y: '0%', duration: 1.2, ease: customEaseIn})
    .fromTo('#e span', {x: '100%'}, {x: '0%', duration: 1.2, ease: customEaseIn}, 0.2)
    .fromTo('#lite span', {x: '-100%'}, {x: '0%', duration: 1.2, ease: customEaseIn}, 0.2)
    .fromTo('#pro span', {x: '100%'}, {x: '0%', duration: 1.2, ease: customEaseIn}, 0.4)
    .fromTo('#services span', {x: '-100%'}, {x: '0%', duration: 1.2, ease: customEaseIn}, 0.4);

  // Services Dropdown Toggle Logic
  const toggle = document.getElementById('serviceDropdownToggle');
  const modal = document.getElementById('servicesModal');
  const input = document.getElementById('selectedServiceInput');

  toggle.addEventListener('click', () => {
    modal.classList.toggle('active');
  });

  document.querySelectorAll('input[name="service"]').forEach(radio => {
    radio.addEventListener('change', (e) => {
      input.value = e.target.value;
      modal.classList.remove('active');
    });
  });

  // Simulated Gmail OTP Logic
  const sendOtpBtn = document.getElementById('sendOtpBtn');
  const otpBox = document.getElementById('otpBox');
  const userEmail = document.getElementById('userEmail');
  const verifyOtpBtn = document.getElementById('verifyOtpBtn');
  const otpInput = document.getElementById('otpInput');
  const otpStatus = document.getElementById('otpStatus');

  sendOtpBtn.addEventListener('click', () => {
    if(userEmail.value && userEmail.value.includes('@gmail.com')) {
      otpBox.classList.add('active');
      otpStatus.textContent = "تم إرسال كود التحقق (6 أرقام) إلى بريدك بنجاح!";
      otpStatus.style.color = "var(--color-neon)";
    } else {
      alert("المرجو إدخال بريد Gmail صحيح أولاً!");
    }
  });

  verifyOtpBtn.addEventListener('click', () => {
    if(otpInput.value.length === 6) {
      otpStatus.textContent = "✓ تم التحقق من البريد الإلكتروني بنجاح!";
      otpStatus.style.color = "#22c55e";
    } else {
      alert("المرجو إدخال الرمز المكون من 6 أرقام بشكل صحيح.");
    }
  });
});

<script type="module">
import * as THREE from "https://cdn.jsdelivr.net/npm/three@0.160.1/build/three.module.js";

var app = document.getElementById("app");
var renderer = new THREE.WebGLRenderer({antialias:false});
renderer.setPixelRatio(Math.min(2, devicePixelRatio||1));
renderer.setSize(innerWidth, innerHeight);
app.appendChild(renderer.domElement);

var scene = new THREE.Scene();
var camera = new THREE.OrthographicCamera(-1,1,1,-1,0,1);

var bgCanvas = document.createElement("canvas");
var bgCtx = bgCanvas.getContext("2d");
var bgTexture = new THREE.CanvasTexture(bgCanvas);

function drawBackground(){
  var w = renderer.domElement.width;
  var h = renderer.domElement.height;
  bgCanvas.width = w;
  bgCanvas.height = h;

  var grd = bgCtx.createLinearGradient(0, 0, w, h);
  grd.addColorStop(0, "#000000");
  grd.addColorStop(0.5, "#05030a");
  grd.addColorStop(1, "#0d0614");
  bgCtx.fillStyle = grd;
  bgCtx.fillRect(0,0,w,h);

  bgCtx.save();
  bgCtx.globalAlpha = 0.2;
  var rg = bgCtx.createRadialGradient(w*0.5, h*0.5, 0, w*0.5, h*0.5, w*0.6);
  rg.addColorStop(0, "rgba(168, 85, 247, 0.4)");
  rg.addColorStop(1, "rgba(0, 0, 0, 0)");
  bgCtx.fillStyle = rg;
  bgCtx.fillRect(0,0,w,h);
  bgCtx.restore();

  bgTexture.needsUpdate = true;
}

drawBackground();

var dropletBuf = new Float32Array(40*4);
var dropletTex = new THREE.DataTexture(dropletBuf, 40, 1, THREE.RGBAFormat, THREE.FloatType);
dropletTex.needsUpdate = true;

var vertSrc = "void main(){ gl_Position = vec4(position, 1.0); }";
var fragSrc = [
"precision highp float;",
"uniform vec2 uRes;",
"uniform sampler2D uBg;",
"void main(){",
"  vec2 uv = gl_FragCoord.xy / uRes;",
"  vec3 col = texture2D(uBg, uv).rgb;",
"  gl_FragColor = vec4(col, 1.0);",
"}"
].join("\n");

var mat = new THREE.ShaderMaterial({
  vertexShader: vertSrc,
  fragmentShader: fragSrc,
  uniforms: {
    uRes: {value: new THREE.Vector2(innerWidth, innerHeight)},
    uBg: {value: bgTexture}
  }
});

scene.add(new THREE.Mesh(new THREE.PlaneGeometry(2,2), mat));

window.addEventListener("resize", function(){
  renderer.setSize(innerWidth, innerHeight);
  mat.uniforms.uRes.value.set(innerWidth, innerHeight);
  drawBackground();
});

(function loop(){
  renderer.render(scene, camera);
  requestAnimationFrame(loop);
})();
</script>
